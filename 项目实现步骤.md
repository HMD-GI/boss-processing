### 步骤 2：Chrome CDP 连接器实现

#### 2.1 核心原理

BOSS直聘的前端使用**字体反爬**——薪资数字通过自定义字体渲染，直接解析 HTML 会得到乱码。但其前端代码本身需要通过内部 API 获取明文数据来渲染：

```
POST /wapi/zpgeek/search/joblist.json
```

该接口返回的 JSON 中包含 `salaryDesc` 字段（如 `30-60K·15薪`），**字体反爬仅在渲染层，API 层为明文**。

**方案**：通过 Chrome DevTools Protocol (CDP) 连接已登录的 Chrome 实例，在页面 JavaScript 上下文中直接调用此 API，复用真实浏览器的 Cookie、指纹和登录态。

#### 2.2 启动带调试端口的 Chrome

```python
# crawler/cdp_connector.py

import subprocess
import os
import time
import json
import websocket  # websocket-client
import requests

class CDPConnector:
    """Chrome CDP 连接管理器"""

    # 隔离 profile 路径（不使用主 Chrome profile，避免冲突和安全风险）
    PROFILE_DIR = os.path.expanduser("~/.boss-zhipin-scraper/chrome-profile")
    CDP_PORT = 9222

    def __init__(self, chrome_path=None, port=9222):
        self.chrome_path = chrome_path or self._detect_chrome_path()
        self.port = port
        self.ws = None
        self.target_id = None
        self.msg_id = 0

    def _detect_chrome_path(self):
        """自动检测 Chrome 可执行文件路径"""
        paths = {
            'win32': [
                r'C:\Program Files\Google\Chrome\Application\chrome.exe',
                r'C:\Program Files (x86)\Google\Chrome\Application\chrome.exe',
            ],
            'darwin': ['/Applications/Google Chrome.app/Contents/MacOS/Google Chrome'],
            'linux': ['google-chrome', 'google-chrome-stable', 'chromium-browser'],
        }
        import platform
        for p in paths.get(platform.system().lower(), []):
            if os.path.exists(p):
                return p
        raise FileNotFoundError("未找到 Chrome，请手动指定路径")

    def start_chrome(self):
        """启动带 CDP 调试端口的 Chrome，使用隔离 profile"""
        os.makedirs(self.PROFILE_DIR, exist_ok=True)
        cmd = [
            self.chrome_path,
            f'--remote-debugging-port={self.port}',
            f'--user-data-dir={self.PROFILE_DIR}',
            '--no-first-run',
            '--no-default-browser-check',
        ]
        subprocess.Popen(cmd, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
        # 等待 CDP 端口就绪
        for _ in range(30):
            try:
                resp = requests.get(f'http://127.0.0.1:{self.port}/json')
                if resp.status_code == 200:
                    break
            except:
                pass
            time.sleep(1)
        else:
            raise RuntimeError("Chrome CDP 端口未就绪")

    def connect(self):
        """连接到 CDP，获取或创建 BOSS直聘页面标签"""
        # 获取已打开的标签页列表
        tabs = requests.get(f'http://127.0.0.1:{self.port}/json').json()
        boss_tab = None
        for tab in tabs:
            if 'zhipin.com' in tab.get('url', ''):
                boss_tab = tab
                break

        if not boss_tab:
            # 创建新标签页并导航到 BOSS直聘
            boss_tab = self._create_tab('https://www.zhipin.com/web/geek/job-recommend')

        self.target_id = boss_tab['id']
        ws_url = boss_tab['webSocketDebuggerUrl']
        self.ws = websocket.create_connection(ws_url)
        return self

    def _create_tab(self, url):
        """创建新标签页"""
        resp = requests.put(
            f'http://127.0.0.1:{self.port}/json/new',
            params={'url': url}
        )
        return resp.json()

    def evaluate(self, expression, await_promise=True):
        """在页面上下文中执行 JavaScript，返回结果"""
        self.msg_id += 1
        msg = {
            'id': self.msg_id,
            'method': 'Runtime.evaluate',
            'params': {
                'expression': expression,
                'returnByValue': True,
                'awaitPromise': await_promise,
            }
        }
        self.ws.send(json.dumps(msg))
        result = json.loads(self.ws.recv())
        return result.get('result', {}).get('result', {}).get('value')

    def close(self):
        """关闭 WebSocket 连接"""
        if self.ws:
            self.ws.close()
```

#### 2.3 首次登录与登录态持久化

```python
# crawler/cdp_connector.py（续）

def setup_chrome(self, login_timeout=300):
    """
    一键启动 Chrome 并等待用户手动登录 BOSS直聘。
    登录态保存在隔离 profile 中，重启后仍保留。
    """
    self.start_chrome()
    self.connect()

    print("请在弹出的 Chrome 窗口中登录 BOSS直聘...")
    print(f"等待登录完成（超时 {login_timeout} 秒）...")

    # 轮询检测登录状态：尝试调用搜索API，看是否返回有效数据
    start = time.time()
    while time.time() - start < login_timeout:
        result = self.evaluate('''
            (async () => {
                try {
                    const resp = await fetch('/wapi/zpgeek/search/joblist.json?keyword=test&page=1', {
                        headers: {'x-requested-with': 'XMLHttpRequest'}
                    });
                    const data = await resp.json();
                    if (data.code === 0 && data.zpData && data.zpData.jobList) {
                        return { logged_in: true, has_salary: !!data.zpData.jobList[0]?.salaryDesc };
                    }
                    return { logged_in: false, code: data.code, message: data.message };
                } catch(e) {
                    return { logged_in: false, error: e.message };
                }
            })()
        ''')

        if result and result.get('logged_in'):
            print("✓ 登录成功！搜索接口可返回明文薪资数据。")
            return True

        # 检测风控码（code:31 环境异常 / code:37 访问频繁）
        if result and result.get('code') in (31, 37):
            print(f"⚠ 检测到风控限制: {result.get('message')}")
            print("  请稍后再试或更换网络环境。")
            return False

        time.sleep(5)  # 每5秒检测一次，避免频繁请求

    print("✗ 登录超时。")
    return False
```

**关键设计要点**（参考 GitHub 项目的经验）：
- **隔离 profile**：使用 `~/.boss-zhipin-scraper/chrome-profile`，不软链接主 Chrome profile，避免锁冲突和安全风险
- **登录态持久化**：登录一次后，登录态保存在隔离 profile 中，后续无需重复登录
- **风控检测**：识别 `code:31`（环境异常）、`code:37`（访问频繁）等风控码，立即停止而非盲目重试

### 步骤 3：职位列表采集

#### 3.1 调用搜索 API 获取明文薪资数据

```python
# crawler/api_fetcher.py

import time
import random
import urllib.parse

class JobListFetcher:
    """BOSS直聘职位列表采集器"""

    SEARCH_API = '/wapi/zpgeek/search/joblist.json'

    def __init__(self, cdp_connector):
        self.cdp = cdp_connector

    def fetch_page(self, keyword, city_code, page, filters=None):
        """
        获取单页职位列表（明文薪资）
        
        参数:
            keyword: 搜索关键词，如 "AI Agent"
            city_code: 城市代码，如 101010100（北京）
            page: 页码，从1开始
            filters: 筛选条件字典（可选）
        """
        params = {
            'scene': '1',
            'query': keyword,
            'city': city_code,
            'experience': '',
            'payType': '',
            'partTime': '',
            'degree': '',
            'industry': '',
            'scale': '',
            'position': '',
            'jobType': '',
            'salary': '',
            'multiBusinessDistrict': '',
            'multiSubway': '',
            'page': str(page),
            'pageSize': '30',
        }

        # 应用筛选条件
        if filters:
            for key, value in filters.items():
                if key in params:
                    params[key] = value

        query_string = urllib.parse.urlencode(params)

        # 在页面上下文中用 XHR 同步调用 API
        # 使用同步 XHR 确保在 CDP evaluate 中能直接拿到结果
        js_code = f'''
            (function() {{
                var xhr = new XMLHttpRequest();
                xhr.open('GET', '{self.SEARCH_API}?{query_string}', false);
                xhr.setRequestHeader('x-requested-with', 'XMLHttpRequest');
                try {{
                    xhr.send();
                    return JSON.parse(xhr.responseText);
                }} catch(e) {{
                    return {{ error: e.message }};
                }}
            }})()
        '''

        result = self.cdp.evaluate(js_code, await_promise=False)
        return result

    def fetch_all(self, keyword, city_code, pages=3, filters=None,
                  delay_range=(12, 22), batch_size=75, batch_delay=(30, 60)):
        """
        多页采集，含反爬限流策略
        
        反爬策略:
        - 每次请求间隔 12~22 秒随机延时
        - 每采集 batch_size 条后休眠 30~60 秒
        - 单次最多 10 页（300条）上限保护
        """
        all_jobs = []
        pages = min(pages, 10)  # 上限保护

        for page in range(1, pages + 1):
            print(f"  正在采集第 {page}/{pages} 页...")

            data = self.fetch_page(keyword, city_code, page, filters)

            if not data or data.get('code') != 0:
                error_msg = data.get('message', '未知错误') if data else '无响应'
                print(f"  ✗ 第 {page} 页采集失败: {error_msg}")
                # 风控码检测
                if data and data.get('code') in (31, 37):
                    print(f"  ⚠ 触发风控，停止采集。")
                    break
                continue

            job_list = data.get('zpData', {}).get('jobList', [])
            if not job_list:
                print(f"  第 {page} 页无更多数据。")
                break

            for job in job_list:
                parsed = self._parse_job(job)
                if parsed:
                    all_jobs.append(parsed)

            print(f"  ✓ 第 {page} 页获取 {len(job_list)} 条，累计 {len(all_jobs)} 条")

            # 每页抓完立即保存（增量写入，Ctrl+C 不丢数据）
            yield all_jobs

            # 限流：页间随机延时
            if page < pages:
                delay = random.uniform(*delay_range)
                print(f"  等待 {delay:.1f} 秒...")
                time.sleep(delay)

                # 批量休眠
                if len(all_jobs) >= batch_size and len(all_jobs) % batch_size == 0:
                    batch_delay_sec = random.uniform(*batch_delay)
                    print(f"  已采集 {len(all_jobs)} 条，休眠 {batch_delay_sec:.1f} 秒...")
                    time.sleep(batch_delay_sec)

        return all_jobs

    def _parse_job(self, raw):
        """解析单条职位数据，提取关键字段"""
        return {
            'job_id': raw.get('encryptJobId', ''),      # 加密职位ID（用于去重和详情页）
            'job_name': raw.get('jobName', ''),
            'salary_desc': raw.get('salaryDesc', ''),    # 明文薪资！如 "30-60K·15薪"
            'city': raw.get('cityName', ''),
            'area': raw.get('businessDistrict', ''),     # 商区
            'experience': raw.get('jobExperience', ''),  # 经验要求
            'education': raw.get('jobDegree', ''),       # 学历要求
            'company_name': raw.get('brandName', ''),
            'company_industry': raw.get('brandIndustry', ''),
            'company_scale': raw.get('brandScaleName', ''),
            'company_stage': raw.get('brandStageName', ''),  # 融资阶段
            'welfare': raw.get('jobLabels', []),          # 职位标签
            'skills': raw.get('skills', []),              # 技能标签
            'boss_name': raw.get('bossName', ''),
            'boss_title': raw.get('bossTitle', ''),
            'boss_active_status': raw.get('bossOnline', False) and '在线' or '离线',
            'security_id': raw.get('securityId', ''),     # 详情页所需上下文
            'lid': raw.get('lid', ''),                    # 详情页所需上下文
            'fetch_time': time.strftime('%Y-%m-%d %H:%M:%S'),
        }
```

### 步骤 4：详情页 JD 抓取

```python
# crawler/detail_fetcher.py

import time
import random

class JobDetailFetcher:
    """职位详情页 JD 抓取"""

    def __init__(self, cdp_connector):
        self.cdp = cdp_connector

    def fetch_detail(self, job):
        """
        抓取单个职位的完整 JD
        
        关键点：
        - 进入详情页时带上列表API返回的 securityId / lid 上下文
        - 只从"职位描述"区域提取JD，不抓取推荐职位等无关内容
        - 若页面出现"登录查看完整内容"，报错并跳过
        """
        job_id = job.get('job_id')
        security_id = job.get('security_id', '')
        lid = job.get('lid', '')

        if not job_id:
            return None

        # 构造详情页URL（带上上下文参数）
        detail_url = f'https://www.zhipin.com/web/geek/job?jobId={job_id}&lid={lid}&securityId={security_id}'

        # 在当前标签页导航到详情页
        self.cdp.evaluate(f'window.location.href = "{detail_url}"', await_promise=False)
        time.sleep(random.uniform(2, 4))  # 等待页面加载

        # 提取JD文本（只从职位描述区域提取）
        js_code = '''
            (function() {
                // 检测登录墙
                if (document.body.innerText.includes('登录查看完整内容')) {
                    return { error: 'login_wall', message: '页面要求登录才能查看完整内容' };
                }

                // 提取职位描述
                var descEl = document.querySelector('.job-detail-section .job-sec-text');
                if (!descEl) {
                    descEl = document.querySelector('[class*="job-sec-text"]');
                }
                var jd = descEl ? descEl.innerText.trim() : '';

                // 提取技能标签
                var skillTags = [];
                document.querySelectorAll('.job-detail-section .tag-list li, .job-tags .tag-item').forEach(function(el) {
                    skillTags.push(el.innerText.trim());
                });

                return { jd: jd, skills: skillTags };
            })()
        '''

        result = self.cdp.evaluate(js_code, await_promise=False)

        if result and result.get('error') == 'login_wall':
            print(f"  ⚠ 职位 {job_id} 触发登录墙，跳过。")
            return None

        if result:
            job['job_desc'] = result.get('jd', '')
            if result.get('skills'):
                existing = set(job.get('skills', []))
                for s in result['skills']:
                    if s and s not in existing:
                        job.setdefault('skills', []).append(s)

        return job
```

### 步骤 5：数据存储与增量写入

```python
# utils/file_io.py

import json
import os
import csv
from datetime import datetime

class DataWriter:
    """数据写入管理器（支持 JSON/CSV，增量写入，按 job_id 去重）"""

    def __init__(self, output_dir=None):
        self.output_dir = output_dir or os.path.expanduser(
            "~/.boss-zhipin-scraper/job-result"
        )
        os.makedirs(self.output_dir, exist_ok=True)

    def save_json(self, jobs, filename=None):
        """保存为 JSON 文件，按 job_id 去重"""
        if not filename:
            timestamp = datetime.now().strftime('%Y%m%d_%H%M')
            filename = f'boss_jobs_{timestamp}.json'
        filepath = os.path.join(self.output_dir, filename)

        # 读取已有数据
        existing = {}
        if os.path.exists(filepath):
            with open(filepath, 'r', encoding='utf-8') as f:
                for job in json.load(f):
                    existing[job.get('job_id')] = job

        # 合并新数据（去重）
        for job in jobs:
            existing[job.get('job_id')] = job  # 同ID则覆盖更新

        merged = list(existing.values())

        with open(filepath, 'w', encoding='utf-8') as f:
            json.dump(merged, f, ensure_ascii=False, indent=2)

        print(f"  ✓ 已保存 {len(merged)} 条到 {filepath}")
        return filepath

    def save_csv(self, jobs, filename=None):
        """保存为 CSV 文件（UTF-8 BOM，Excel 兼容）"""
        if not filename:
            timestamp = datetime.now().strftime('%Y%m%d_%H%M')
            filename = f'boss_jobs_{timestamp}.csv'
        filepath = os.path.join(self.output_dir, filename)

        if not jobs:
            return filepath

        fields = ['job_name', 'salary_desc', 'city', 'area', 'experience',
                  'education', 'company_name', 'company_industry', 'company_scale',
                  'skills', 'job_desc', 'boss_name', 'boss_active_status',
                  'fetch_time']

        with open(filepath, 'w', newline='', encoding='utf-8-sig') as f:
            writer = csv.DictWriter(f, fieldnames=fields, extrasaction='ignore')
            writer.writeheader()
            for job in jobs:
                # 列表字段转为字符串
                row = dict(job)
                if isinstance(row.get('skills'), list):
                    row['skills'] = ' | '.join(row['skills'])
                writer.writerow(row)

        print(f"  ✓ 已保存 CSV 到 {filepath}")
        return filepath

    def merge_files(self, file1, file2, output):
        """合并两个 JSON 文件，按 job_id 去重"""
        data1 = self._load_json(file1)
        data2 = self._load_json(file2)

        merged = {}
        for job in data1 + data2:
            merged[job.get('job_id')] = job

        result = list(merged.values())
        with open(output, 'w', encoding='utf-8') as f:
            json.dump(result, f, ensure_ascii=False, indent=2)

        print(f"✓ 合并完成: {len(data1)} + {len(data2)} → {len(result)} 条 (去重后)")
        return output

    def _load_json(self, filepath):
        if not os.path.exists(filepath):
            return []
        with open(filepath, 'r', encoding='utf-8') as f:
            return json.load(f)
```

    conn = CDPConnector(port=args.cdp_port)
    conn.connect()
    result = conn.evaluate('''
        (async () => {
            try {
                const resp = await fetch('/wapi/zpgeek/search/joblist.json?keyword=test&page=1', {
                    headers: {'x-requested-with': 'XMLHttpRequest'}
                });
                const data = await resp.json();
                return { code: data.code, has_jobs: !!(data.zpData?.jobList?.length) };
            } catch(e) { return { error: e.message }; }
        })()
    ''')
    if result and result.get('code') == 0:
        print("✓ BOSS直聘已登录，搜索接口正常")
    else:
        print(f"✗ 登录态异常: {result}")
    conn.close()

def cmd_crawl(args):
    """执行采集任务"""
    conn = CDPConnector(port=args.cdp_port)
    conn.connect()

    city_code = get_city_code(args.city)
    if not city_code:
        print(f"✗ 未知城市: {args.city}")
        print("使用 --list-cities 查看支持的城市列表")
        return

    fetcher = JobListFetcher(conn)
    writer = DataWriter(args.output)

    # 构建筛选条件
    filters = {}
    if args.scale: filters['scale'] = args.scale
    if args.salary: filters['salary'] = args.salary
    if args.experience: filters['experience'] = args.experience
    if args.degree: filters['degree'] = args.degree

    print(f"开始采集: 关键词='{args.keyword}', 城市={args.city}, 页数={args.pages}")

    all_jobs = []
    for batch in fetcher.fetch_all(args.keyword, city_code, args.pages, filters):
        all_jobs = batch
        # 每页抓完立即保存（增量写入）
        writer.save_json(all_jobs)

    # 详情页抓取
    if not args.no_detail and all_jobs:
        print(f"\n开始抓取详情页 JD ({len(all_jobs)} 条)...")
        detail_fetcher = JobDetailFetcher(conn)
        for i, job in enumerate(all_jobs):
            print(f"  [{i+1}/{len(all_jobs)}] {job['job_name']} - {job['company_name']}")
            detail_fetcher.fetch_detail(job)
            writer.save_json(all_jobs)  # 每条详情抓完立即保存
            if i < len(all_jobs) - 1:
                import time, random
                time.sleep(random.uniform(3, 6))  # 详情页间延时

    # 最终保存
    output_file = writer.save_json(all_jobs)
    if args.format == 'csv':
        writer.save_csv(all_jobs)

    print(f"\n✓ 采集完成！共 {len(all_jobs)} 条职位数据")
    print(f"  结果文件: {output_file}")

    conn.close()

def main():
    parser = argparse.ArgumentParser(description='BOSS直聘数据采集工具')

```python
# backend/api.py

from fastapi import FastAPI, HTTPException, BackgroundTasks
from pydantic import BaseModel
from typing import Optional, List
from sqlalchemy import create_engine, func, desc
from sqlalchemy.orm import sessionmaker
from models.job import Job, CrawlLog, Base
import os

app = FastAPI(title="BOSS直聘数据采集系统", version="1.0")

# 数据库连接
DB_URL = os.getenv('DB_URL', 'mysql+pymysql://root:root@localhost:3306/boss_db')
engine = create_engine(DB_URL, pool_size=10, max_overflow=20)
SessionLocal = sessionmaker(bind=engine)

# 自动建表
Base.metadata.create_all(engine)

# --- 请求模型 ---

class CrawlRequest(BaseModel):
    keywords: List[str]
    city: str = "上海"
    pages: int = 3
    fetch_detail: bool = True

# --- API 端点 ---

@app.get("/api/jobs")
def list_jobs(
    keyword: Optional[str] = None,
    city: Optional[str] = None,
    salary_min: Optional[int] = None,
    salary_max: Optional[int] = None,
    page: int = 1,
    page_size: int = 20,
    sort_by: str = "fetch_time",
    sort_order: str = "desc"
):
    """分页查询职位列表，支持多条件筛选"""
    db = SessionLocal()
    query = db.query(Job)

    if keyword:
        query = query.filter(Job.job_name.contains(keyword))
    if city:
        query = query.filter(Job.city == city)

    # 排序
    sort_column = getattr(Job, sort_by, Job.fetch_time)
    if sort_order == "desc":
        query = query.order_by(desc(sort_column))
    else:
        query = query.order_by(sort_column)

    total = query.count()
    jobs = query.offset((page - 1) * page_size).limit(page_size).all()

    db.close()
    return {
        "total": total,
        "page": page,
        "page_size": page_size,
        "data": [_job_to_dict(j) for j in jobs]
    }

@app.get("/api/jobs/{job_id}")
def get_job_detail(job_id: str):
    """获取单条职位详情"""
    db = SessionLocal()
    job = db.query(Job).filter(Job.job_id == job_id).first()
    db.close()
    if not job:
        raise HTTPException(status_code=404, detail="职位不存在")
    return _job_to_dict(job, full=True)

@app.post("/api/crawl/start")
def start_crawl(req: CrawlRequest, background_tasks: BackgroundTasks):
    """触发爬取任务（异步执行）"""
    background_tasks.add_task(_run_crawl, req)
    return {"status": "started", "message": f"已启动采集任务: {req.keywords}"}

@app.get("/api/crawl/status")
def crawl_status():
    """查询最近采集任务状态"""
    db = SessionLocal()
    logs = db.query(CrawlLog).order_by(desc(CrawlLog.timestamp)).limit(20).all()
    db.close()
    return {"recent_logs": [_log_to_dict(l) for l in logs]}

@app.get("/api/stats")
def stats():
    """数据统计（总数、城市分布、薪资分布等）"""
    db = SessionLocal()

    total = db.query(func.count(Job.id)).scalar()

    # 城市分布
    city_dist = db.query(
        Job.city, func.count(Job.id)
    ).group_by(Job.city).order_by(desc(func.count(Job.id))).all()

    # 薪资分布（简单按区间统计）
    salary_stats = db.query(
        func.min(Job.salary_desc),
        func.max(Job.salary_desc)
    ).first()

    db.close()
    return {
        "total_jobs": total,
        "city_distribution": [{"city": c, "count": n} for c, n in city_dist],
    }

# --- 辅助函数 ---

def _job_to_dict(job, full=False):
    d = {
        "job_id": job.job_id,
        "job_name": job.job_name,
        "salary_desc": job.salary_desc,
        "city": job.city,
        "company_name": job.company_name,
        "experience": job.experience,
        "education": job.education,
    }
    if full:
        d.update({
            "job_desc": job.job_desc,
            "area": job.area,
            "company_industry": job.company_industry,
            "company_scale": job.company_scale,
            "skills": job.skills,
            "welfare": job.welfare,
            "boss_name": job.boss_name,
            "boss_active_status": job.boss_active_status,
            "fetch_time": job.fetch_time.isoformat() if job.fetch_time else None,
        })
    return d

def _log_to_dict(log):
    return {
        "keyword": log.keyword,
        "city": log.city,
        "page": log.page,
        "fetched_count": log.fetched_count,
        "status": log.status,
        "error_msg": log.error_msg,
        "timestamp": log.timestamp.isoformat() if log.timestamp else None,
    }

def _run_crawl(req: CrawlRequest):
    """后台执行采集任务"""
    # 这里调用阶段一的采集引擎
    from crawler.cdp_connector import CDPConnector
    from crawler.api_fetcher import JobListFetcher
    from crawler.detail_fetcher import JobDetailFetcher
    from utils.city_codes import get_city_code

    conn = CDPConnector()
    conn.connect()
    city_code = get_city_code(req.city)
    fetcher = JobListFetcher(conn)
    db = SessionLocal()

    for keyword in req.keywords:
        for batch in fetcher.fetch_all(keyword, city_code, req.pages):
            # 写入数据库（去重：job_id 存在则更新）
            for job_data in batch:
                existing = db.query(Job).filter(Job.job_id == job_data['job_id']).first()
                if existing:
                    # 更新已有记录
                    for key, value in job_data.items():
                        setattr(existing, key, value)
                else:
                    db.add(Job(**job_data))
            db.commit()

            # 记录日志
            log = CrawlLog(keyword=keyword, city=req.city, fetched_count=len(batch))
            db.add(log)
            db.commit()

    db.close()
    conn.close()
```

## 反爬策略总结

| 反爬措施 | 应对策略 | 实现方式 |
|----------|----------|----------|
| **字体反爬** | 直接调用内部 API `/wapi/zpgeek/search/joblist.json`，获取明文 `salaryDesc` | CDP 在页面上下文中执行 XHR |
| **请求频率限制** | 翻页间隔 12~22 秒随机延时；每 75 条休眠 30~60 秒 | `time.sleep(random.uniform(12, 22))` |
| **单次上限** | 单次最多 10 页（300条） | `pages = min(pages, 10)` |
| **登录态校验** | 使用隔离 Chrome profile 持久化登录态 | `--user-data-dir` 隔离目录 |
| **风控检测** | 识别 `code:31`/`code:37` 风控码，立即停止 | API 响应码检查 |
| **行为分析** | API 请求从页面内发起，与正常浏览行为一致 | 页面上下文 XHR 调用 |
| **DOM 降级** | 默认禁用 DOM 提取，避免字体反爬导致薪资乱码 | 仅 API 模式，`--allow-dom-fallback` 可选 |
| **登录墙** | 详情页检测"登录查看完整内容"，报错跳过 | 页面文本检测 |
